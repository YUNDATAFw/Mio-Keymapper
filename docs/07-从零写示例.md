## 7.2 完整版 autoAttack（生产可直接用）

这是当前插件工程里的真实脚本，逐段注释：

```java
package mio.shortcutKey        // 模块归属，与 config.json 的 service name 对应

import mio.utils.MioEngFun     // 只用到核心模块

private String ACTION_ID = "autoAttack"   // 快捷键 id，唯一

// keyInit：SHORTCUT_KEY 服务线程启动后第一个调用
export fn Void keyInit() {
    // 注册一个 SWITCH 型快捷键：标题「自动攻击」，id=autoAttack，分组「战斗」
    MioEngFun.onKeyInit("自动攻击", "SWITCH", ACTION_ID,
        "开启后以配置频率持续按下Q键进行连续攻击（任意模式可用）", "战斗")
}

// onTriggerDown：SWITCH 拨到「开」时进入，阻塞运行直到拨回「关」
export fn Void onTriggerDown() {

    // ① 注入通道守卫：没有可用通道直接退出，不给用户弹错误
    if(!MioEngFun.isEngineReady()) {
        MioEngFun.showToast("按键注入未就绪，请先开启输入法或Shizuku")
        return
    }

    Int intervalTime = 25    // 每次点按间隔
    Int pressDuration = 5    // 单次按下时长

    // ② 触发主循环：只要开关仍是「开」就继续
    while(MioEngFun.isTriggerDown(ACTION_ID)) {

        // ③ 单击 Q 键
        Bool ok = MioEngFun.tap("KEYCODE_Q", pressDuration)

        // ④ 注入失败（例如通道中途断掉）→ 提示并停止
        if(!ok) {
            MioEngFun.showToast("按键注入失败，已停止")
            return
        }

        // ⑤ 间隔
        MioEngFun.sleep(intervalTime)

        // ⑥ 检测到松开/关掉 → 跳出循环（onTriggerDown 返回）
        if(MioEngFun.isTriggerUp(ACTION_ID)) {
            break
        }
    }
}
```

### 为什么这样写

- **`while(isTriggerDown)` 而非 `while(true)`**：宿主把触发状态写进状态表，开关关闭即退出并返回，交由宿主收尾。
- **通道守卫与失败即停**：`tap` 返回 `false` 表示真的注入失败，继续空转只浪费电量。
- **`return` vs `break` 的区别**：`return` 提前退出（本轮不视为正经"结束"），宿主此时若检测到按键仍按住，**不会**发 `onTriggerUp`；`break` 走正常结束路径，触发 `onTriggerUp`。

## 7.3 综合示例（多快捷键 + 收尾 + 浮窗 + 传感器）

同一模块可以注册多个快捷键，宿主有 `SHORTCUT_KEY` 服务就按 `keyInit → trigger` 循环跑。下面示例演示：按钮型连发、开关型脚本、`onTriggerUp` 收尾、`MioFloating` 提示、传感器辅助。

```java
package mio.shortcutKey

import mio.utils.MioEngFun
import mio.utils.MioGamepad
import mio.utils.MioSensor
import mio.utils.MioFloating

private String ID_FIRE = "combat"
private String ID_AUTOPILOT = "autopilot"

export fn Void keyInit() {
    // 按钮型：按住期间连发
    MioEngFun.onKeyInit("连发",   "BUTTON", ID_FIRE,     "按住持续点按攻击键")
    // 开关型：开→自动走路，关→停下
    MioEngFun.onKeyInit("自动走路","SWITCH", ID_AUTOPILOT, "持续按住前进")
}

// ========== 按钮：按住连发 ==========
export fn Void onTriggerDown() {
    if(!MioEngFun.isEngineReady()) {
        MioEngFun.showToast("注入通道未就绪")
        return
    }
    while(MioEngFun.isTriggerDown(ID_FIRE)
       || MioEngFun.isTriggerDown(ID_AUTOPILOT)) {   // 两个格子共用同一处理循环
        // 连发
        if(MioEngFun.isTriggerDown(ID_FIRE)) {
            MioEngFun.tap("KEYCODE_J", 5)
        }
        // 自动走路：按住W
        if(MioEngFun.isTriggerDown(ID_AUTOPILOT)) {
            MioEngFun.keyDown("KEYCODE_W")
        } else {
            MioEngFun.keyUp("KEYCODE_W")
        }
        MioEngFun.sleep(20)
        // 都停了才退出
        if(MioEngFun.isTriggerUp(ID_FIRE) && MioEngFun.isTriggerUp(ID_AUTOPILOT)) {
            MioEngFun.keyUp("KEYCODE_W")   // 收尾：确保松开
            break
        }
    }
}

// 按钮专属：按下瞬间的在手提示
export fn Void onButtonPress() {
    MioFloating.showToast("连发已启动")
}
export fn Void onButtonRelease() {
    MioFloating.showToast("连发已停止")
}

// 开关专属：开关切换提示
export fn Void onSwitchOn()  { MioFloating.showToast("自动走路 开") }
export fn Void onSwitchOff() { MioFloating.showToast("自动走路 关") }

// 通用收尾：某些一次性动作可放在这（本示例没用到，留作模式示范）
export fn Void onTriggerUp() {
    MioEngFun.keyUp("KEYCODE_W")
}
```

## 7.5 命令宏：把指令发进游戏聊天栏

真实工程的 `mio/shortcutKey/mc/MioMcCmd.dsl`——按键注入、文本注入、配置读取三合一，被全插件 24 个指令类快捷键复用：

```java
package mio.shortcutKey.mc

import mio.utils.MioEngFun
// 读取宿主已保存配置（HostData 存储，未保存时为 null）
private fn Object readCfg(String key) = call("HostData.get")

// 执行聊天栏指令：打开聊天框 → 等待 → 输入指令 → 回车发送
export fn Void runCommand(String actionId, String text) {

    // 防止其他快捷键触发串台
    if(!MioEngFun.isTriggerDown(actionId)) {
        return
    }

    // 检查注入引擎是否就绪
    if(!MioEngFun.isEngineReady()) {
        MioEngFun.showToast("按键注入未就绪，请先开启输入法或Shizuku")
        return
    }

    // 指令时序从配置页读取（未保存时默认 20/300/200）
    Int tapMs = 20
    Object c1 = readCfg("cmd.tapMs")
    if(c1 != null) {
        tapMs = c1
    }
    Int openMs = 300
    Object c2 = readCfg("cmd.openMs")
    if(c2 != null) {
        openMs = c2
    }
    Int sendMs = 200
    Object c3 = readCfg("cmd.sendMs")
    if(c3 != null) {
        sendMs = c3
    }
    // 打开聊天框（按住 / 键 20ms）
    Bool opened = MioEngFun.tap("KEYCODE_SLASH", tapMs)
    if(!opened) {
        MioEngFun.showToast("按键注入失败，已停止")
        return
    }

    // 等待聊天框打开
    MioEngFun.sleep(openMs)

    // 输入指令内容
    Bool typed = MioEngFun.inputText(text)
    if(!typed) {
        MioEngFun.showToast("文本注入失败，已停止")
        return
    }

    // 等待输入完成
    MioEngFun.sleep(sendMs)

    // 回车发送
    MioEngFun.tap("KEYCODE_ENTER", tapMs)
}
```

要点：

- **`isTriggerDown(actionId)` 串台守卫**：`runCommand` 是公共模块，被多个快捷键共用——确认是**本 id 触发的这次调用**才继续。
- **三段时序全部可配**（`cmd.tapMs`/`cmd.openMs`/`cmd.sendMs`，对应 §7.7 的配置页全局 Tab）——不同机型/网络下聊天框打开速度不同，把时序交给用户调。
- 调用方（如 `timeDay.dsl`）只需一行：`MioMcCmd.runCommand(ACTION_ID, "time set day")`。
- 这也是「`inputText` 依赖输入法通道」的典型场景：Shizuku 通道下文本注入可能表现不同，失败时靠返回值分支提示。

## 7.6 构建、安装、验证

1. 编译：`java -jar dsl-cli.jar build <插件工程目录>`（或走设备端控制台流程）→ 产物 `build/app.signed.zip`。
2. 安装：Mio 插件页面选择该 ZIP 安装；有敏感权限则跟随授权。
3. 重启浮窗服务（确保 `loadDslPlugins` 重新加载）→ 悬浮快捷键条上应出现你注册的标题。
4. 测试：
   - BUTTON：按住格子 → 观察注入；松开 → `onTriggerUp/onButtonRelease` 收尾。
   - SWITCH：点一次动作开始，再点一次动作结束。
   - 出错看通知栏：见 [08 调试与错误处理](08-调试与错误处理.md)。

验证点对应的行为：

| 预期行为 | 检查 |
|---|---|
| 按住就循环注入，松开立即停 | `isTriggerDown` + `break` 逻辑、宿主 `isTriggerUp` 复位 |
| 注入通道未就绪时报提示而非崩溃 | `isEngineReady` 守卫 + 提前 `return` |
| 收尾动作只发生一次 | `onTriggerUp` / 模型专属松开（仅真实松开才触发） |

## 7.7 加配置页（把写死的参数交给用户）

`autoAttack` 里 `intervalTime = 25` 是写死的——改成用户可调要两步：

**① `res/config.json` 加一条声明**

```json
"service": [
  { "name": "main",   "tag": "MAINSERVICE" },
  { "name": "config", "tag": "CONFIG" },
  { "name": "mio.shortcutKey.autoAttack", "tag": "SHORTCUT_KEY" }
]
```

**② 根目录 `src/script/config.dsl` 写 `schema()`**

```java
import mio.utils.MioConfig

export fn Void schema() {
    MioConfig.onSection("自动攻击")
    MioConfig.onSwitch("autoAttack.enabled", "注册快捷键", "关闭并保存后重启插件，该快捷键不再注册", true)
    MioConfig.onText("autoAttack.interval", "点按间隔（毫秒）", "", "25", "5~500", "int", 5, 500)
}
```

**③ 业务模块读回**（`autoAttack.dsl` 开头加一行包装，循环里用配置值）

```java
private fn Object readCfg(String key) = call("HostData.get")

export fn Void onTriggerDown() {
    if(!MioEngFun.isEngineReady()) { MioEngFun.showToast("注入未就绪"); return }

    Int intervalTime = 25
    Object iv = readCfg("autoAttack.interval")   // 未保存时为 null
    if(iv != null) { intervalTime = iv }

    // …原有 while 循环，用 intervalTime…
}
```

**④ 开关控制注册**（`keyInit` 里）

```java
export fn Void keyInit() {
    Object en = readCfg("autoAttack.enabled")
    if(en == false) { return }              // 用户关掉并保存 → 重启后不再注册
    MioEngFun.onKeyInit("自动攻击", "SWITCH", ACTION_ID, "…", "战斗")
}
```

保存配置 = 宿主落盘 → **自动重启插件** → 上面两处读到新值。控件总表、校验矩阵、`onButton` 回调见 [11 配置页与 HostData](11-配置页与HostData.md)。

**⑤ 完整合并版**（上面三段合成一个可直接复制的文件，已通过 `dsl-cli build` 验证）：

```java
package mio.shortcutKey

import mio.utils.MioEngFun

private String ACTION_ID = "autoAttack"

// 读取宿主已保存配置（HostData 存储，未保存时为 null）
private fn Object readCfg(String key) = call("HostData.get")

export fn Void keyInit() {
    // 开关配置：用户关闭并保存 → 重启插件即不再注册
    Object en = readCfg("autoAttack.enabled")
    if(en == false) {
        return
    }
    MioEngFun.onKeyInit("自动攻击", "SWITCH", ACTION_ID,
        "开启后以配置频率持续按下Q键进行连续攻击（任意模式可用）", "战斗")
}

export fn Void onTriggerDown() {
    if(!MioEngFun.isEngineReady()) {
        MioEngFun.showToast("按键注入未就绪，请先开启输入法或Shizuku")
        return
    }

    // 数值配置：未保存时用默认值
    Int intervalTime = 25
    Object iv = readCfg("autoAttack.interval")
    if(iv != null) {
        intervalTime = iv
    }
    Int pressDuration = 5

    while(MioEngFun.isTriggerDown(ACTION_ID)) {
        Bool ok = MioEngFun.tap("KEYCODE_Q", pressDuration)
        if(!ok) {
            MioEngFun.showToast("按键注入失败，已停止")
            return
        }
        MioEngFun.sleep(intervalTime)
        if(MioEngFun.isTriggerUp(ACTION_ID)) {
            break
        }
    }
}
```

对应的 `config.dsl` 用 ② 的 `schema()` 即可（key 对上：`autoAttack.enabled` / `autoAttack.interval`）。

package mio.ws

import mio.ws.MioWs

export fn Void serverInit() {
